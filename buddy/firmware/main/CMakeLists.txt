set(buddy_panel_requires esp_lcd esp_lcd_st77916)
# Only the NV3023 build links the controller driver for the 240x240 module.
if(CONFIG_BUDDY_PANEL_NV3023_240)
    list(APPEND buddy_panel_requires esp_lcd_nv3023)
endif()

idf_component_register(
    SRCS "main.c"
         "buddy_state.c"
         "buddy_font.c"
         "buddy_display.c"
         "buddy_ui.c"
         "buddy_protocol.c"
         "buddy_touch.c"
         "buddy_i2c.c"
         "buddy_sound.c"
         "buddy_pet.c"
    INCLUDE_DIRS "."
    REQUIRES esp_driver_gpio esp_driver_ledc esp_driver_spi esp_driver_i2c
             ${buddy_panel_requires} esp_driver_usb_serial_jtag esp_driver_i2s
             nvs_flash esp_timer esp_psram cjson mbedtls
)

# Packed 24x24 CJK fallback (U+4E00-U+9FFF) copied from iot-agentos, plus the
# generated punctuation/full-width extension.
target_add_binary_data(${COMPONENT_LIB} "cjk24_cjk.bin" BINARY)
target_add_binary_data(${COMPONENT_LIB} "cjk24_ext.bin" BINARY)
# Five 16 kHz stereo PCM notification sounds, generated by tools/generate_sounds.py.
target_add_binary_data(${COMPONENT_LIB} "sounds.pcm" BINARY)
